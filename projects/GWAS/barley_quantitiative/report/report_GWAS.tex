	\section{Summary}
		
		\textbf{About}
		\begin{itemize}\spacing
			\item I run GAPIT on the quantitative barley traits and the genotype data from a barley population
			\item I also implemented the core functionality of a GWAS, expecially for understanding the linear regression of the markers
		\end{itemize}
		
		\textbf{Data}
		\begin{itemize}\spacing
			\item This data was provided by the MLU Halle during my master programme 
			\item Quantitative phenotype traits and a genotype matrix for multiple breed lines and multiple markers of Chromosome 2 in barely
			\item[] \begin{tabular}{lll}
					HEA &= Heading date                		&(Ährenschieben) \\
					TGW &= Thousand Grain Weights      		&(Tausendkorngewicht) \\
					HEI &= Height                      		&(Höhe) \\
					RIP &= Ripening time               		&(Reifedauer) \\
					PM  &= Powdery Mildew Susceptibility   	&(Mehltauafälligkeit)
					\end{tabular}
		\end{itemize}

		
		\textbf{Quantitative Traits (Phenotype)}
		\begin{itemize}\spacing
			\item Normally distribution of HEA, TGW, HEI and RIP (potential poly genetic orchestration)
			\item Bimodal distribution of PM, hint for a genetically distinct groups in the population 
			\item Strong anti correlation between HEA and RIP
			\item Weak correlation between TGW and RIP
			\item Weak correlation between TGW and HEA
		\end{itemize}

			\item If the desired trait is a greater croop (yields more food), one can clearly see that TGW (Thousand Grain Weight) slightly correlates with the RIP (Ripening Time)
			\item The crop might have more time accumulating consumable nutrients (sugars, proteins, lipids)
			\item The scatter also shows that one would select crops that ripen later
		\end{itemize}
	
		TGW $\times$ HEA
		\begin{itemize}\spacing
			\item TGW slightly anti-correlates with HEA (Heading Date)

		\end{itemize}
		
		HEA $\times$ RIP
		\begin{itemize}\spacing
			\item Strong anti correlation
			\item High chance of a co-regulated gene orchestration

			\item Maybe not relevant for breeding, but interesting behavior of barley, when they have weaker resistance against powder mildew, the crops heights reduce varaince and shrink

				\item Meaning that no marker is surprisingly significant
			\end{itemize}
			
			\textbf{HEI}
			\begin{itemize}
				\item Both model outputs are similiar (aligns with QQ)

				\item Same region as GLM
			\end{itemize}
			
			\textbf{RIP}
			\begin{itemize}
				\item Both model outputs are similiar (aligns with QQ)
				\item FarmCPU identifies 2 Maker to be highly significant with a high $R^{2}$ (positional close)

				\item These markers could correspond to the anti correlation found previously
				\item Same region as GLM
			\end{itemize}
			
			\textbf{PM}
			\begin{itemize}
				\item Both models outputs identical (aligns with QQ)
				\item One can clearly see that  a wider genome region $(0.10-0.15 cM$) has a genetic pattern for the powdery mildew susceptibility (with ok'ish significance $p\approx0.05$, but not corrected)
				\item One could dig into this pattern, since it might align with the phenotype distribution (not relevant for this report)
			\end{itemize}

		\subsection{Significant Marker}
			
			\textbf{Linkage Disequilibrium (LD)}
			
				\begin{figure}[h!]
					\centering
					\includegraphics[scale=0.5]{"./LD/LD_significant_marker.png"}
				\end{figure}
				
				\begin{itemize}
					\item All significant marker pairs show a $LD > 0.8$, meaning that they are non-random associated markers (co-regulate traits)
					\item This implies that a selection of an quantitative trait, based on one of these significant markers, also triggers another traits 		
					\item This also implies that the effect are additive
				\end{itemize}
				
			
			\textbf{Phenotype Distribution - HEA}

		\section{Marker-assisted selection (trait prediction)}
		
			\textbf{Formula:}
			\begin{center}
				\boxed{	y = \mu + \alpha_{marker} \cdot SNP + \epsilon}
			\end{center}
			
 			\begin{itemize}
 				\item $\mu$ = Phenotype (A)
 				\item $\alpha_{marker}$ = Effect at Marker
 				\item SNP = Genotype at SNP (or Marker)
 				\item Error
 			\end{itemize}
