	\section{Summary}
	
		\paragraph{ABF2 and ABF3 play major role in increased cell division and transcriptional activity as reponse to nitrate stress}
	
		\begin{itemize}
			\item This findings supports the fact, that nitrogen is key signal in cell growth and absorbs nitrate at the root
			\item The GOEA suggests that ABF2 and ABF3 share major functionality but also divide there tasks, coordinating the response cascade 
		\end{itemize}
	
		\paragraph{Shared roles}
			\begin{itemize}
				\item Both seem to down regulate the maintenance program of the chromosome and upregulating RNA synthesis
				\item Both seems to up regulate cell division, RNA programs and down regulating cell shaping processes (actin formation)
				\item Both acting on ATP regulation
				\item Both acting on enzymes: Down regulating Isomerase, Dow regulating Nucleases, Up regulating helicases
			\end{itemize}
		
		\paragraph{Different roles} \
		
			ABF2		
			\begin{itemize}
				\item Down regulates Chromatin, 
			\end{itemize}
			
			ABF§ 
			\begin{itemize}
				\item Down regulates Microtubuli, up regulating Peptidase, down regulating environmental stimuli
			\end{itemize}
			
		\paragraph{Cooridnated Roles}
			\begin{itemize}
				\item ABF2 seems to stronger act on the down regulation of Heterochromatin, whereas ABF3 acts on the upregulation of euchromatin
			\end{itemize}
